const tabButtons = document.querySelectorAll('.tab-button');
const homeTab = document.getElementById('home-tab');
const chatTab = document.getElementById('chat-tab');
const chatBack = document.getElementById('chat-back');

tabButtons.forEach((btn) => {

chatBack.addEventListener('click', () => switchTab('home'));

function switchTab(tabName) {
  activeTab = tabName;
  homeTab.hidden = tabName !== 'home';
  chatTab.hidden = tabName !== 'chat';
  tabButtons.forEach((btn) => {

    btn.classList.toggle('active', isActive);

  });
}

- [ ] **Step 1: Add stream id in `SidebarProvider.ts`**

```typescript
private currentChatStreamId = 0;

case 'chat-ask': {
  const query = message.query;
  if (!query?.trim()) return;
  const streamId = ++this.currentChatStreamId;
  webview.postMessage({ type: 'chat-user-turn', text: query, streamId });

  await this.chatService.askQuestion(query, {
    onStageChange: (stage, meta) => {
      if (streamId !== this.currentChatStreamId) return;   // stale stream — drop
      webview.postMessage({ type: 'chat-stage', stage, noteCount: meta?.noteCount, streamId });
    },
    onChunk: (text) => {
      if (streamId !== this.currentChatStreamId) return;
      webview.postMessage({ type: 'chat-token', text, streamId });
    },
    onDone: (finalText, citations) => {
      if (streamId !== this.currentChatStreamId) return;
      webview.postMessage({ type: 'chat-done', finalText, citations, streamId });
    },
    onError: (err) => {
      if (streamId !== this.currentChatStreamId) return;
      webview.postMessage({ type: 'chat-error', message: err.message, streamId });
    },
  });
  return;
}

case 'chat-stop': {
  this.currentChatStreamId++;   // bumps id; any in-flight callbacks for the prior id will be dropped
  webview.postMessage({ type: 'chat-stopped' });
  return;
}
```
